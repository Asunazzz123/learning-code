semaphore size = N;  // 缓冲区容量
semaphore mutex1 = 1; // 消费者互斥
semaphore num = 0 // 缓冲区产品数
semaphore mutex2 = 1  // 生产者互斥
consumer(){
    P(mutex1);
    int i = 0; // 已经获得的产品量
    while(i<K){
        get();
        V(size);
        P(num);
        i++;
    }
    V(mutex1);
}


producer(){
    while(true){
        P(mutex2); // 生产者互斥
        P(size);
        put();
        V(num);
        V(mutex2); // 退出临界区后释放
    }
}
