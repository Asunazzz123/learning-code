semaphore mutex = 1 // 通道互斥
semaphore inside = C // 允许进入量

visitor(){
    // 进入通道
    P(inside);
    P(mutex);
    in();
    V(mutex);
    // 参观
    visit();
    // 离开通道
    P(mutex);
    out();
    V(mutex);
    V(inside);
}
